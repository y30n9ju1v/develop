== 1. 토크나이저

[source,lean]
----
-- Compiler/Token.lean
namespace Nand2Lean.Compiler

inductive Token where
  | kwFunction | kwVar | kwIf | kwElse | kwWhile | kwReturn | kwLet
  | lparen | rparen | lbrace | rbrace
  | semi | comma
  | plus | minus | star | eq | lt | gt | assign
  | intConst (n : Nat)
  | identifier (s : String)
  deriving DecidableEq, Repr

def keywords : List (String × Token) :=
  [("function", .kwFunction), ("var", .kwVar), ("if", .kwIf), ("else", .kwElse),
   ("while", .kwWhile), ("return", .kwReturn), ("let", .kwLet)]

partial def tokenize (s : String) : List Token :=
  go s.toList
where
  go : List Char → List Token
    | [] => []
    | c :: cs =>
        if c.isWhitespace then go cs
        else if c.isDigit then
          let (digits, rest) := (c :: cs).span Char.isDigit
          .intConst (String.mk digits).toNat! :: go rest
        else if c.isAlpha then
          let (chars, rest) := (c :: cs).span (fun ch => ch.isAlphanum || ch = '_')
          let word := String.mk chars
          (keywords.lookup word |>.getD (.identifier word)) :: go rest
        else
          match c, cs with
          | '(', _ => .lparen :: go cs
          | ')', _ => .rparen :: go cs
          | '{', _ => .lbrace :: go cs
          | '}', _ => .rbrace :: go cs
          | ';', _ => .semi :: go cs
          | ',', _ => .comma :: go cs
          | '+', _ => .plus :: go cs
          | '-', _ => .minus :: go cs
          | '*', _ => .star :: go cs
          | '=', '=' :: rest => .eq :: go rest
          | '=', _ => .assign :: go cs
          | '<', _ => .lt :: go cs
          | '>', _ => .gt :: go cs
          | '/', '/' :: rest => go (rest.dropWhile (· ≠ '\n'))  -- 줄 주석
          | _, _ => go cs  -- 알 수 없는 문자는 무시 (실전에서는 에러로 처리)

end Nand2Lean.Compiler
----

[TIP]
====
*확인*: `#eval Nand2Lean.Compiler.tokenize "let x = 3 + 4;"`를 실행하면 `[.kwLet, .identifier "x", .assign, .intConst 3, .plus, .intConst 4, .semi]`가 나와야 합니다.
====

== 2. AST

[source,lean]
----
-- Compiler/Ast.lean
namespace Nand2Lean.Compiler

inductive BinOp where | add | sub | mul | eq | lt | gt deriving Repr

inductive Expr where
  | intLit (n : Nat)
  | varRef (name : String)
  | binOp (op : BinOp) (l r : Expr)
  | call (name : String) (args : List Expr)
  deriving Repr

inductive Stmt where
  | letStmt (name : String) (value : Expr)
  | ifStmt (cond : Expr) (thenBody : List Stmt) (elseBody : List Stmt)
  | whileStmt (cond : Expr) (body : List Stmt)
  | returnStmt (value : Expr)
  deriving Repr

structure FunctionDecl where
  name : String
  params : List String
  locals : List String
  body : List Stmt
  deriving Repr

end Nand2Lean.Compiler
----

== 3. 재귀 하강 파서

Lean4에서 "실패할 수 있고 위치를 소비하는" 파서는 `StateT (List Token) Option`으로 표현하는 것이 가장 단순합니다.

[source,lean]
----
-- Compiler/Parser.lean
import Compiler.Ast

namespace Nand2Lean.Compiler

abbrev ParserM := StateT (List Token) Option

def peek : ParserM (Option Token) := do
  match (← get) with
  | [] => pure none
  | t :: _ => pure (some t)

def advance : ParserM Token := do
  match (← get) with
  | [] => failure
  | t :: rest => set rest; pure t

def expect (t : Token) : ParserM Unit := do
  let t' ← advance
  if t' = t then pure () else failure

partial def parsePrimary : ParserM Expr := do
  match ← advance with
  | .intConst n => pure (.intLit n)
  | .identifier name =>
      match ← peek with
      | some .lparen =>
          expect .lparen
          let args ← parseArgs
          expect .rparen
          pure (.call name args)
      | _ => pure (.varRef name)
  | .lparen =>
      let e ← parseExpr
      expect .rparen
      pure e
  | _ => failure
where
  parseArgs : ParserM (List Expr) := do
    match ← peek with
    | some .rparen => pure []
    | _ =>
        let first ← parseExpr
        let rest ← parseArgsRest
        pure (first :: rest)
  parseArgsRest : ParserM (List Expr) := do
    match ← peek with
    | some .comma => expect .comma; let e ← parseExpr; let rest ← parseArgsRest; pure (e :: rest)
    | _ => pure []

partial def parseTerm : ParserM Expr := do
  let first ← parsePrimary
  parseTermRest first
where
  parseTermRest (acc : Expr) : ParserM Expr := do
    match ← peek with
    | some .star => expect .star; let r ← parsePrimary; parseTermRest (.binOp .mul acc r)
    | _ => pure acc

partial def parseArith : ParserM Expr := do
  let first ← parseTerm
  parseArithRest first
where
  parseArithRest (acc : Expr) : ParserM Expr := do
    match ← peek with
    | some .plus => expect .plus; let r ← parseTerm; parseArithRest (.binOp .add acc r)
    | some .minus => expect .minus; let r ← parseTerm; parseArithRest (.binOp .sub acc r)
    | _ => pure acc

partial def parseExpr : ParserM Expr := do
  let l ← parseArith
  match ← peek with
  | some .eq => expect .eq; let r ← parseArith; pure (.binOp .eq l r)
  | some .lt => expect .lt; let r ← parseArith; pure (.binOp .lt l r)
  | some .gt => expect .gt; let r ← parseArith; pure (.binOp .gt l r)
  | _ => pure l

end Nand2Lean.Compiler
----

`parseTerm`(곱셈)이 `parseArith`(덧셈/뺄셈)보다 먼저, `parseArith`가 비교 연산자보다 먼저 오는 3단 구조가 연산자 우선순위(`*` > `+`/`-` > 비교)를 인코딩합니다 — 이건 6~7장에서 이미 봤던 "우선순위를 함수 호출 순서로 인코딩한다"는 패턴을 파서에도 그대로 적용한 것입니다.

== 4. 문장과 함수

[source,lean]
----
namespace Nand2Lean.Compiler

partial def parseStmt : ParserM Stmt := do
  match ← peek with
  | some .kwLet =>
      expect .kwLet
      let name ← advance
      match name with
      | .identifier n =>
          expect .assign
          let e ← parseExpr
          expect .semi
          pure (.letStmt n e)
      | _ => failure
  | some .kwIf =>
      expect .kwIf; expect .lparen
      let cond ← parseExpr
      expect .rparen; expect .lbrace
      let thenBody ← parseStmts
      expect .rbrace
      let elseBody ← do
        match ← peek with
        | some .kwElse =>
            expect .kwElse; expect .lbrace
            let b ← parseStmts
            expect .rbrace
            pure b
        | _ => pure []
      pure (.ifStmt cond thenBody elseBody)
  | some .kwWhile =>
      expect .kwWhile; expect .lparen
      let cond ← parseExpr
      expect .rparen; expect .lbrace
      let body ← parseStmts
      expect .rbrace
      pure (.whileStmt cond body)
  | some .kwReturn =>
      expect .kwReturn
      let e ← parseExpr
      expect .semi
      pure (.returnStmt e)
  | _ => failure
where
  parseStmts : ParserM (List Stmt) := do
    match ← peek with
    | some .kwLet | some .kwIf | some .kwWhile | some .kwReturn =>
        let s ← parseStmt
        let rest ← parseStmts
        pure (s :: rest)
    | _ => pure []

def parseIdentList : ParserM (List String) := do
  match ← peek with
  | some .rparen => pure []
  | _ =>
      match ← advance with
      | .identifier n =>
          let rest ← do
            match ← peek with
            | some .comma => expect .comma; parseIdentList
            | _ => pure []
          pure (n :: rest)
      | _ => failure

def parseFunction : ParserM FunctionDecl := do
  expect .kwFunction
  let nameT ← advance
  let name := match nameT with | .identifier n => n | _ => "?"
  expect .lparen
  let params ← parseIdentList
  expect .rparen
  expect .lbrace
  let locals ← parseVarDecls
  let body ← parseStmt.parseStmts
  expect .rbrace
  pure { name, params, locals, body }
where
  parseVarDecls : ParserM (List String) := do
    match ← peek with
    | some .kwVar =>
        expect .kwVar
        let nameT ← advance
        let n := match nameT with | .identifier n => n | _ => "?"
        expect .semi
        let rest ← parseVarDecls
        pure (n :: rest)
    | _ => pure []

def parseProgram (tokens : List Token) : Option FunctionDecl :=
  (parseFunction.run tokens).map Prod.fst

end Nand2Lean.Compiler
----

== 5. 실제 소스 파일 파싱해보기

[source,lean]
----
-- CompilerMain.lean
import Compiler.Parser

def sampleSource : String :=
"function fact(n) {
  var result;
  let result = 1;
  while (n > 1) {
    let result = result * n;
    let n = n - 1;
  }
  return result;
}"

def main : IO Unit := do
  let tokens := Nand2Lean.Compiler.tokenize sampleSource
  match Nand2Lean.Compiler.parseProgram tokens with
  | none => IO.eprintln "parse error"
  | some decl => IO.println (toString (repr decl))
----

[source,bash]
----
lake build compilermain   -- lakefile.lean에 등록 후
lake exe compilermain
----

기대되는 출력(정확한 포맷은 `Repr` 인스턴스에 따라 다르지만, 구조는 이렇게 나와야 합니다):

[source,text]
----
{ name := "fact",
  params := ["n"],
  locals := ["result"],
  body := [.letStmt "result" (.intLit 1),
           .whileStmt (.binOp .gt (.varRef "n") (.intLit 1))
             [.letStmt "result" (.binOp .mul (.varRef "result") (.varRef "n")),
              .letStmt "n" (.binOp .sub (.varRef "n") (.intLit 1))],
           .returnStmt (.varRef "result")] }
----

[TIP]
====
*확인*: `body` 리스트에 `whileStmt`가 있고, 그 안에 곱셈(`.mul`)과 뺄셈(`.sub`)이 올바른 순서로 중첩되어 있으면 파서가 정확히 동작한 것입니다. 흔한 실패 지점은 `while` 본문 파싱 후 `rbrace`를 소비하지 않아 다음 파싱이 어긋나는 경우입니다 — 에러 메시지 없이 `parse error`만 나온다면, `parseStmt`의 각 갈래에서 `expect .rbrace`가 빠지지 않았는지 확인하세요.
====

== 6. 심볼 테이블: 매개변수와 지역 변수의 인덱스

[source,lean]
----
-- Compiler/SymbolTable.lean
namespace Nand2Lean.Compiler

inductive VarKind where | param | local_ deriving Repr

structure VarInfo where
  kind : VarKind
  idx : Nat

def buildSymbolTable (decl : FunctionDecl) : List (String × VarInfo) :=
  (decl.params.zipIdx.map (fun (n, i) => (n, ⟨.param, i⟩))) ++
  (decl.locals.zipIdx.map (fun (n, i) => (n, ⟨.local_, i⟩)))

end Nand2Lean.Compiler
----

`VarInfo.idx`가 10장에서 그대로 VM `argument`/`local` 세그먼트의 인덱스가 됩니다. `fact` 예제라면 `buildSymbolTable`의 결과는 `[("n", ⟨.param, 0⟩), ("result", ⟨.local_, 0⟩)]`이어야 합니다.

== 이 장에서 확정한 것

* 토크나이저, AST, 3단 우선순위 재귀 하강 파서를 처음부터 끝까지 실행 가능한 코드로 완성했습니다.
* 실제 소스 문자열(`fact` 함수)을 토큰화하고 파싱해, 곱셈과 뺄셈이 올바른 우선순위와 중첩 구조로 AST에 반영됨을 확인했습니다.
* 매개변수/지역 변수를 인덱스와 함께 심볼 테이블로 정리해, 10장의 세그먼트 인덱스로 바로 이어질 준비를 마쳤습니다.

다음 장에서는 이 AST와 심볼 테이블을 받아 실제로 7~8장의 VM 명령어를 생성하는 코드 생성기를 완성하고, `fact` 함수를 실제로 컴파일해 어셈블리까지 끝까지 뽑아봅니다.
