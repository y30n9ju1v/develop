== 시나리오: 약간의 높이 변화가 있는 루프

12장의 정사각형 루프에 높이(`z`) 변화를 더해, 순수 2D가 아닌 진짜 3D 궤적을 만듭니다.

[source,python]
----
import numpy as np

def skew(w):
    return np.array([[0, -w[2], w[1]], [w[2], 0, -w[0]], [-w[1], w[0], 0]])

def exp_so3(w):
    theta = np.linalg.norm(w)
    if theta < 1e-8:
        return np.eye(3) + skew(w)
    axis = w / theta
    K = skew(axis)
    return np.eye(3) + np.sin(theta) * K + (1 - np.cos(theta)) * (K @ K)

def log_so3(R):
    cos_theta = np.clip((np.trace(R) - 1) / 2, -1, 1)
    theta = np.arccos(cos_theta)
    if theta < 1e-8:
        return np.array([R[2, 1] - R[1, 2], R[0, 2] - R[2, 0], R[1, 0] - R[0, 1]]) / 2
    w_hat = (R - R.T) / (2 * np.sin(theta))
    w_axis = np.array([w_hat[2, 1], w_hat[0, 2], w_hat[1, 0]])
    return w_axis * theta

def left_jacobian_so3(w):
    theta = np.linalg.norm(w)
    K = skew(w)
    if theta < 1e-8:
        return np.eye(3) + 0.5 * K
    A = (1 - np.cos(theta)) / theta ** 2
    B = (theta - np.sin(theta)) / theta ** 3
    return np.eye(3) + A * K + B * (K @ K)

def exp_se3(xi):
    w, v = xi[:3], xi[3:]
    T = np.eye(4)
    T[:3, :3] = exp_so3(w)
    T[:3, 3] = left_jacobian_so3(w) @ v
    return T

def log_se3(T):
    R, t = T[:3, :3], T[:3, 3]
    w = log_so3(R)
    v = np.linalg.solve(left_jacobian_so3(w), t)
    return np.concatenate([w, v])

def se3_inv(T):
    R, t = T[:3, :3], T[:3, 3]
    T_inv = np.eye(4)
    T_inv[:3, :3] = R.T
    T_inv[:3, 3] = -R.T @ t
    return T_inv

def boxplus_se3(T, xi):
    return T @ exp_se3(xi)

def make_pose(axis, angle, t):
    T = np.eye(4)
    T[:3, :3] = exp_so3(np.array(axis) * angle)
    T[:3, 3] = t
    return T

gt = [np.eye(4)]
gt.append(make_pose([0, 0, 1], np.pi / 2, [2, 0, 0.3]))
gt.append(make_pose([0, 0, 1], np.pi, [2, 2, 0.6]))
gt.append(make_pose([0, 0, 1], -np.pi / 2, [0, 2, 0.3]))
----

로봇이 정사각형을 돌면서 절반 지점(`gt[2]`)까지 `0.6m` 상승했다가 다시 `gt[0]`(`z=0`) 근처로 내려오는, 순수 2D 그래프 최적화로는 다룰 수 없는 궤적입니다.

== 오도메트리 엣지: IMU 회전 + 이동 센서 위치

각 구간(`0→1, 1→2, 2→3, 3→0`)마다, 회전은 13장의 자이로 프리인테그레이션으로, 위치는 별도의 이동 센서(예: 휠 오도메트리)로 얻는다고 가정합니다 — 실제 로봇에서 흔한 구성입니다(IMU는 회전에 강하고, 순수 가속도 이중적분은 드리프트가 매우 커서 별도의 위치 센서를 함께 쓰는 경우가 많습니다).

[source,python]
----
def preintegrate_gyro(gyro, dt):
    dR = np.eye(3)
    for w in gyro:
        dR = dR @ exp_so3(w * dt)
    return dR

rng = np.random.default_rng(77)
edges = [(0, 1), (1, 2), (2, 3), (3, 0)]
dt, T_seg = 0.01, 1.0
Nk = int(T_seg / dt)

gyro_noise_std = 0.003
odom_noise_std = 0.03
loop_noise_std = 0.05

Z = {}
for (i, j) in edges:
    T_ij_true = se3_inv(gt[i]) @ gt[j]
    w_const = log_so3(T_ij_true[:3, :3]) / T_seg    # 구간 동안 일정한 각속도라고 가정
    gyro = np.tile(w_const, (Nk, 1)) + rng.standard_normal((Nk, 3)) * gyro_noise_std
    dR = preintegrate_gyro(gyro, dt)

    noise_std = loop_noise_std if (i, j) == (3, 0) else odom_noise_std
    t_noisy = T_ij_true[:3, 3] + rng.standard_normal(3) * noise_std

    T_meas = np.eye(4)
    T_meas[:3, :3] = dR
    T_meas[:3, 3] = t_noisy
    Z[(i, j)] = T_meas
----

`(3, 0)` 엣지는 루프 클로저입니다 — 회전은 여전히 자이로에서 얻지만, 위치는 오도메트리 누적 오차가 큰 상태에서 다시 시작점 근처를 인식한 별도의 관측(더 큰 노이즈)이라고 가정합니다.

== 포즈 그래프 최적화: 12장의 코드를 그대로 재사용

잔차, 야코비안, 가우스-뉴턴 루프는 12장과 완전히 동일합니다 — 이 장의 새로운 부분은 오직 "측정값 `Z_ij`가 어디서 왔는가"(IMU + 오도메트리)뿐입니다. 이게 리 군/리 대수라는 공통 언어의 힘입니다 — 측정값의 출처가 바뀌어도 최적화 코드는 손댈 필요가 없습니다.

[source,python]
----
def residual(T_i, T_j, Z_ij):
    return log_se3(se3_inv(Z_ij) @ (se3_inv(T_i) @ T_j))

def build_residuals(est):
    r = []
    for (i, j) in edges:
        r.append(residual(est[i], est[j], Z[(i, j)]))
    return np.concatenate(r)

def numeric_jacobian(est, eps=1e-6):
    r0 = build_residuals(est)
    J = np.zeros((len(r0), 18))
    for pose_idx in range(1, 4):
        for k in range(6):
            d = np.zeros(6)
            d[k] = eps
            est_plus = est.copy()
            est_plus[pose_idx] = boxplus_se3(est[pose_idx], d)
            est_minus = est.copy()
            est_minus[pose_idx] = boxplus_se3(est[pose_idx], -d)
            J[:, (pose_idx - 1) * 6 + k] = (
                (build_residuals(est_plus) - build_residuals(est_minus)) / (2 * eps)
            )
    return J

est = [gt[0]] + [gt[k] @ exp_se3(rng.standard_normal(6) * 0.1) for k in range(1, 4)]
errors = []
for iteration in range(10):
    r = build_residuals(est)
    errors.append(np.linalg.norm(r))
    J = numeric_jacobian(est)
    H = J.T @ J + 1e-6 * np.eye(18)
    dx = np.linalg.solve(H, -J.T @ r)
    for pose_idx in range(1, 4):
        est[pose_idx] = boxplus_se3(est[pose_idx], dx[(pose_idx - 1) * 6:(pose_idx - 1) * 6 + 6])

print([round(e, 4) for e in errors])
----

== 이 책이 보여주고자 한 것

1장에서 오일러각의 짐벌락으로 시작해서, 여기까지 오일러각은 단 한 번도 다시 등장하지 않았습니다. 회전 행렬과 쿼터니언(2, 3장)이라는 두 가지 특이점 없는 표현을 확인하고, 그 공통된 대수적 뼈대(리 군, 리 대수)를 뽑아내(4장), 회전과 각속도를 정확하고 특이점 없이 오가는 지수/로그 사상을 만들었습니다(5, 6장). 이걸 이동까지 포함하는 `SE(3)`으로 확장하고(7, 8장), 좌표계를 바꿔가며 값을 옮기는 애드조인트(9장)를 갖춘 뒤, 이 모든 걸 최적화 문제에 연결하는 `⊞`/`⊟`와 야코비안(10, 11장)을 도입했습니다. 마지막 세 장(12~14장)은 이 도구들이 실제로 포즈 그래프 최적화와 IMU 프리인테그레이션이라는, 로봇공학에서 매일 쓰이는 알고리즘의 정확한 수학적 기초라는 것을 직접 구현하고 검증하며 확인했습니다.

== 요약

* IMU 자이로 프리인테그레이션(13장)으로 얻은 회전과 별도 이동 센서로 얻은 위치를 결합해 `SE(3)` 오도메트리 엣지를 만들고, 12장과 동일한 포즈 그래프 최적화 코드로 노이즈 낀 3D 궤적을 실제 궤적에 가깝게 복원했습니다.
* 궤적에 높이 변화(`z`)를 더한 진짜 3D 문제였는데도, `SE(3)`이라는 공통 표현 덕분에 최적화 코드 자체는 전혀 바뀌지 않았습니다 — 이것이 이 책 전체가 리 군/리 대수라는 공통 언어를 다진 이유입니다.
* 루프 클로저의 노이즈가 커지면 최적화 결과의 정확도가 함께 나빠지며, 실전에서는 측정값마다 다른 신뢰도(정보 행렬 가중치)를 반영하는 것이 다음 단계로 필요합니다.
* 이 장을 끝으로, 1장의 짐벌락 문제에서 시작한 여정이 특이점 없는 3D 자세 추정 파이프라인으로 완성되었습니다.
