9장 끝에서 "여러 사용자가 서로 다른 데이터를 봐야 할 때는 서버가 상태를 기억해야 한다"고 예고했습니다. 로그인이 정확히 그 경우입니다 — 서버는 "이 요청을 보낸 사람이 누구인지"를 요청 자체만으로는 알 수 없고, 어딘가에 저장해둔 정보를 참조해야 합니다. 이 장에서는 쿠키에 저장한 세션 ID로 로그인 상태를 유지하는 가장 기본적인 인증을 구현합니다.

== 비밀번호는 절대 평문으로 저장하지 않는다

사용자의 비밀번호를 데이터베이스에 그대로 저장하면, 데이터베이스가 유출됐을 때 모든 사용자의 비밀번호가 그대로 노출됩니다. `bcrypt`는 비밀번호를 단방향으로 해싱해, 원래 비밀번호를 복원할 수 없는 형태로 저장합니다.

[source,bash]
----
go get golang.org/x/crypto/bcrypt
----

[source,go]
----
package main

import "golang.org/x/crypto/bcrypt"

func hashPassword(password string) (string, error) {
	b, err := bcrypt.GenerateFromPassword([]byte(password), bcrypt.DefaultCost)
	return string(b), err
}

func checkPassword(hash, password string) bool {
	return bcrypt.CompareHashAndPassword([]byte(hash), []byte(password)) == nil
}
----

`hashPassword`가 회원가입 시 비밀번호를 해싱해 저장할 때, `checkPassword`가 로그인 시 입력된 비밀번호가 저장된 해시와 일치하는지 확인할 때 쓰입니다. `bcrypt`는 같은 비밀번호를 해싱해도 매번 다른 결과를 내지만(내부에 무작위 salt가 섞이므로), `CompareHashAndPassword`가 이를 올바르게 비교합니다.

== 세션 저장소: 세션 ID와 사용자를 연결하기

세션 ID(추측할 수 없는 무작위 문자열)를 사용자 ID에 매핑하는 저장소를 만듭니다. 여기서는 메모리 맵으로 구현하지만, 서버가 여러 대라면 Redis 같은 공유 저장소가 필요합니다(이 책의 범위를 넘어갑니다).

[source,go]
----
package main

import (
	"crypto/rand"
	"encoding/hex"
	"sync"
)

type SessionStore struct {
	mu       sync.Mutex
	sessions map[string]int64 // sessionID -> userID
}

func NewSessionStore() *SessionStore {
	return &SessionStore{sessions: make(map[string]int64)}
}

func newSessionID() string {
	b := make([]byte, 16)
	rand.Read(b)
	return hex.EncodeToString(b)
}

func (s *SessionStore) Create(userID int64) string {
	id := newSessionID()
	s.mu.Lock()
	s.sessions[id] = userID
	s.mu.Unlock()
	return id
}

func (s *SessionStore) UserID(sessionID string) (int64, bool) {
	s.mu.Lock()
	defer s.mu.Unlock()
	uid, ok := s.sessions[sessionID]
	return uid, ok
}
----

`crypto/rand`(암호학적으로 안전한 난수 생성기, `math/rand`가 아닙니다)로 세션 ID를 만드는 게 중요합니다 — 예측 가능한 값이면 공격자가 다른 사용자의 세션 ID를 추측해 로그인 상태를 가로챌 수 있습니다. `sync.Mutex`는 여러 요청이 동시에 세션 맵을 읽고 쓸 때(Go 서버는 각 요청을 별도 고루틴에서 처리하므로) 데이터 경쟁을 막습니다.

== 쿠키로 세션 ID 전달하기

[source,go]
----
import "net/http"

const sessionCookieName = "session_id"

func setSessionCookie(w http.ResponseWriter, sessionID string) {
	http.SetCookie(w, &http.Cookie{
		Name:     sessionCookieName,
		Value:    sessionID,
		Path:     "/",
		HttpOnly: true,
		SameSite: http.SameSiteLaxMode,
	})
}

func currentUserID(r *http.Request, store *SessionStore) (int64, bool) {
	cookie, err := r.Cookie(sessionCookieName)
	if err != nil {
		return 0, false
	}
	return store.UserID(cookie.Value)
}
----

`HttpOnly: true`는 자바스크립트가 이 쿠키 값을 읽지 못하게 막습니다 — XSS 취약점이 있더라도 공격자가 세션 쿠키를 훔쳐가는 걸 어렵게 하는 방어입니다(3장에서 templ의 자동 이스케이프로 XSS 자체를 막는 것과는 다른 층위의 방어입니다).

== 인증 미들웨어: 로그인하지 않은 접근을 막기

2장에서 배운 미들웨어 패턴을 그대로 씁니다 — 특정 라우트를 감싸, 세션이 없으면 로그인 페이지로 리다이렉트합니다.

[source,go]
----
func requireAuth(store *SessionStore, next http.HandlerFunc) http.HandlerFunc {
	return func(w http.ResponseWriter, r *http.Request) {
		if _, ok := currentUserID(r, store); !ok {
			http.Redirect(w, r, "/login", http.StatusSeeOther)
			return
		}
		next(w, r)
	}
}
----

[source,go]
----
mux.HandleFunc("GET /dashboard", requireAuth(store, func(w http.ResponseWriter, r *http.Request) {
	uid, _ := currentUserID(r, store)
	fmt.Fprintf(w, "<p>user %d의 대시보드</p>", uid)
}))
----

== 로그인 전후의 접근 흐름을 httptest로 검증

브라우저 없이도, 쿠키 저장소(`cookiejar`)를 가진 HTTP 클라이언트로 실제 로그인 흐름 전체를 테스트할 수 있습니다.

[source,go]
----
func readBody(resp *http.Response) string {
	b, _ := io.ReadAll(resp.Body)
	return string(b)
}

func TestLoginFlow(t *testing.T) {
	store := NewSessionStore()
	hashed, _ := hashPassword("secret123")

	mux := http.NewServeMux()
	mux.HandleFunc("GET /login", func(w http.ResponseWriter, r *http.Request) {
		fmt.Fprint(w, "<form>...로그인 폼...</form>")
	})
	mux.HandleFunc("POST /login", func(w http.ResponseWriter, r *http.Request) {
		r.ParseForm()
		if checkPassword(hashed, r.FormValue("password")) {
			sid := store.Create(42)
			setSessionCookie(w, sid)
			fmt.Fprint(w, "<p>로그인 성공</p>")
			return
		}
		w.WriteHeader(http.StatusUnauthorized)
		fmt.Fprint(w, "<p>로그인 실패</p>")
	})
	mux.HandleFunc("GET /dashboard", requireAuth(store, func(w http.ResponseWriter, r *http.Request) {
		uid, _ := currentUserID(r, store)
		fmt.Fprintf(w, "<p>user %d의 대시보드</p>", uid)
	}))

	srv := httptest.NewServer(mux)
	defer srv.Close()

	jar, _ := cookiejar.New(nil)
	client := &http.Client{Jar: jar}   // 쿠키를 자동으로 저장하고 다음 요청에 실어 보낸다

	resp, _ := client.Get(srv.URL + "/dashboard")
	t.Logf("로그인 전: 최종 경로 = %s, 상태 코드 = %d", resp.Request.URL.Path, resp.StatusCode)

	resp2, _ := client.PostForm(srv.URL+"/login", url.Values{"password": {"secret123"}})
	t.Logf("로그인 응답: %s", readBody(resp2))

	resp3, _ := client.Get(srv.URL + "/dashboard")
	t.Logf("로그인 후: %s", readBody(resp3))
}
----

[source]
----
로그인 전: 최종 경로 = /login, 상태 코드 = 200
로그인 응답: <p>로그인 성공</p>
로그인 후: <p>user 42의 대시보드</p>
----

세 가지가 정확히 확인됩니다. 첫째, 로그인 전에 `/dashboard`에 접근하면 `requireAuth`가 `/login`으로 리다이렉트해, 클라이언트가 최종적으로 도착한 경로가 `/login`입니다. 둘째, 올바른 비밀번호로 로그인하면 세션이 생성되고 쿠키가 클라이언트에 저장됩니다(`cookiejar`가 이 쿠키를 기억합니다). 셋째, 그 이후 같은 클라이언트로 `/dashboard`에 접근하면 이번엔 리다이렉트 없이 `user 42의 대시보드`가 바로 응답됩니다 — 쿠키에 담긴 세션 ID로 서버가 "이 요청은 42번 사용자의 것"이라는 걸 정확히 알아냈다는 뜻입니다.

== 직접 해보기

* 로그아웃 핸들러(`POST /logout`)를 만들어, 세션 쿠키를 읽어 `SessionStore`에서 해당 세션을 제거하고, 쿠키의 만료 시간을 과거로 설정해 브라우저에서도 삭제되도록 만들어보세요.

.해답 보기
[%collapsible]
====
[source,go]
----
func (s *SessionStore) Destroy(sessionID string) {
	s.mu.Lock()
	delete(s.sessions, sessionID)
	s.mu.Unlock()
}

mux.HandleFunc("POST /logout", func(w http.ResponseWriter, r *http.Request) {
	cookie, err := r.Cookie(sessionCookieName)
	if err == nil {
		store.Destroy(cookie.Value)
	}
	http.SetCookie(w, &http.Cookie{
		Name:    sessionCookieName,
		Value:   "",
		Path:    "/",
		Expires: time.Unix(0, 0),   // 과거 시각으로 설정하면 브라우저가 즉시 삭제
		MaxAge:  -1,
	})
	http.Redirect(w, r, "/login", http.StatusSeeOther)
})
----

서버 쪽(`SessionStore.Destroy`)과 클라이언트 쪽(쿠키 만료) 둘 다 처리하는 게 중요합니다 — 서버 쪽만 지우면 브라우저가 여전히 (이제는 무효한) 세션 ID를 계속 보내고, 클라이언트 쿠키만 지우면 서버 메모리에 죽은 세션이 계속 쌓입니다.

== 요약

* 비밀번호는 `bcrypt`로 해싱해 저장하고, 세션 ID는 `crypto/rand`로 예측 불가능하게 생성해 쿠키로 클라이언트에 전달합니다.
* `HttpOnly` 쿠키는 자바스크립트가 세션 값을 읽지 못하게 막아, XSS가 발생하더라도 세션 탈취를 어렵게 하는 추가 방어층입니다.
* `requireAuth` 미들웨어가 세션이 없는 요청을 로그인 페이지로 리다이렉트하며, 이는 2장의 미들웨어 패턴을 그대로 재사용한 것입니다.
* `cookiejar`를 가진 HTTP 클라이언트로 로그인 전/후의 접근 제어 전체 흐름을 브라우저 없이 검증할 수 있습니다.
* 다음 장에서는 세션으로 식별한 사용자에게 서버가 먼저 데이터를 밀어 보내는 실시간 갱신(SSE)을 다룹니다.
